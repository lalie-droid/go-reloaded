// ReglePonctuation gère les ponctuations selon les règles spécifiées.
func ReglePonctuation(line string) string {
	// Liste des ponctuations à gérer
	punctuations := []string{".", ",", "!", "?", ":", ";", "'"}
	var result strings.Builder
	i := 0
	for i < len(line) {
		if contains(punctuations, string(line[i])) {
			// Vérifie si c'est un groupe de ponctuation (..., !?, etc.)
			if i+1 < len(line) && contains(punctuations, string(line[i+1])) {
				// C'est un groupe de ponctuation
				j := i + 1
				for j < len(line) && contains(punctuations, string(line[j])) {
					j++
				}
				group := line[i:j]
				// Coller le groupe au nextMot précédent
				if i > 0 && line[i-1] != ' ' {
					result.WriteByte(' ')
				}
				result.WriteString(group)
				// Ajouter un espace après le groupe si nécessaire
				if j < len(line) && line[j] != ' ' {
					result.WriteByte(' ')
				}
				i = j
			} else {
				// Ponctuation simple
				if i > 0 && line[i-1] != ' ' {
					result.WriteByte(' ')
				}
				result.WriteByte(line[i])
				// Ajouter un espace après la ponctuation si nécessaire
				if i+1 < len(line) && line[i+1] != ' ' && !unicode.IsSpace(rune(line[i+1])) {
					result.WriteByte(' ')
				}
				i++
			}
		} else {
			result.WriteByte(line[i])
			i++
		}
	}
	return result.String()
}

func handleApostrophes(line string) string {
	var result strings.Builder
	i := 0
	for i < len(line) {
		if line[i] == '\'' {
			// Trouve la deuxième apostrophe
			j := i + 1
			for j < len(line) && line[j] != '\'' {
				j++
			}
			if j < len(line) {
				// Extrait le texte entre les apostrophes
				word := strings.TrimSpace(line[i+1 : j])
				// Reconstruit avec les apostrophes collées
				result.WriteString("'" + word + "'")
				i = j + 1
			} else {
				// Pas de deuxième apostrophe, on garde l'apostrophe telle quelle
				result.WriteByte(line[i])
				i++
			}
		} else {
			result.WriteByte(line[i])
			i++
		}
	}
	return result.String()
}

// ADevientAn transformation de "a" en "an" 

func ADevientAn(line string) string {
	tabMots := strings.Fields(line)
	for i := 0; i < len(tabMots)-1; i++ {
		if strings.ToLower(tabMots[i]) == "a" {       // on peut juste faire tabMot = a OU A
			nextMot := tabMots[i+1]
			if startsWithVowelOrH(nextMot) {
				tabMots[i] = "an"
			}
		}
	}
	return strings.Join(tabMots, " ")
}


// startsWithVowelOrH vérifie si un nextMot commence par une voyelle ou un 'h'.
func startsWithVowelOrH(word string) bool {
	if len(word) == 0 {
		return false
	}
	firstChar := unicode.ToLower(rune(word[0]))
	return strings.ContainsRune("aeiouh", firstChar)
}

// contains vérifie si une slice contient une valeur.
func contains(slice []string, val string) bool {
	for _, item := range slice {
		if item == val {
			return true
		}
	}
	return false
}

// max retourne le maximum de deux entiers.
func max(a, b int) int {
	if a > b {
		return a
	}
	return b
}
