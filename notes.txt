package fonction

import "fmt"

func Testmot() {
	// Définir le mot à chercher
	motAChercher := "it"

	// Appeler la fonction MotRecherche
	trouve, err := fonctions.MotRecherche(motAChercher)
	if err != nil {
		fmt.Println("Erreur :", err)
		return
	}

	// Afficher le résultat
	if trouve {
		fmt.Printf("✅ Le mot '%s' a été trouvé dans le fichier sample.txt.\n", motAChercher)
	} else {
		fmt.Printf("❌ Le mot '%s' n'a pas été trouvé dans le fichier sample.txt.\n", motAChercher)
	}
}
